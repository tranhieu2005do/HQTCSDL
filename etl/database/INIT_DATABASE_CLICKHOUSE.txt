CREATE DATABASE football;

SHOW DATABASES;

USE football;

CREATE TABLE football.dim_team
(
    id UInt32,
    name String,
    code String,
    founded UInt16,
    logo_url String
)
ENGINE = MergeTree()
ORDER BY id; -- ngầm định id là primary key

CREATE TABLE football.dim_venue
(
    id UInt32,
    name String,
    address Nullable(String),
    city Nullable(String),
    capacity Nullable(UInt32),
    image_url String
)
ENGINE = MergeTree()
ORDER BY id;

CREATE TABLE dim_players
(
    id UInt32,
    name String,
    first_name Nullable(String),
    last_name Nullable(String),
    date_birth Nullable(Date),
    nationality Nullable(String),
    height_cm Nullable(UInt16),
    weight_kg Nullable(UInt16),
    image_url String
)
ENGINE = MergeTree()
ORDER BY id;

CREATE TABLE fact_player_statistic
(
    player_id UInt32,
    team_id UInt32,

    position LowCardinality(String),

    appearences UInt16,
    lineups UInt16,
    minutes UInt32,

    total_shots UInt16,
    total_goals UInt16,
    total_assists UInt16,
    total_saves UInt16,
    total_passes UInt32,

    yellow_cards UInt8,
    yellowred_cards UInt8,
    red_cards UInt8
)
ENGINE = MergeTree()
ORDER BY (team_id, player_id);

CREATE TABLE fact_fixture
(
    id UInt32,

    referee LowCardinality(String),

    date Date,

    venue_id UInt32,

    round UInt8,

    home_id UInt32,
    away_id UInt32,

    home_goals UInt8,
    away_goals UInt8
)
ENGINE = MergeTree()
ORDER BY (date, id);

CREATE TABLE dim_coaches
(
    id UInt32,

    name String,

    photo_url String
)
ENGINE = MergeTree()
ORDER BY id;

CREATE TABLE fact_lineups
(
    id UInt32,

    fixture_id UInt32,
    team_id UInt32,
    coach_id UInt32,

    formation LowCardinality(String)
)
ENGINE = MergeTree()
ORDER BY (fixture_id, team_id);

CREATE TABLE fact_fixture_statistics
(
    id UInt32,

    fixture_id UInt32,
    team_id UInt32,

    shots_on_goal UInt8,
    shots_off_goal UInt8,
    total_shots UInt8,
    blocked_shots UInt8,

    shots_inside_box UInt8,
    shots_outside_box UInt8,

    fouls UInt8,
    corner_kicks UInt8,
    offsides UInt8,

    possession UInt8,

    yellow_cards UInt8,
    red_cards UInt8,

    goalkeeper_saves UInt8,

    total_passes UInt16,
    accurate_passes UInt16,

    pass_accuracy UInt8,

    expected_goals Float32
)
ENGINE = MergeTree()
ORDER BY (fixture_id, team_id);

CREATE TABLE fact_fixture_player_statistics
(
    id UInt32,

    fixture_id UInt32,
    team_id UInt32,
    player_id UInt32,

    minutes_played UInt16,

    rating Decimal(3,1),

    substitute Bool,

    shots UInt8,
    goals UInt8,
    assists UInt8,
    saves UInt8,

    passes_total UInt16,
    passes_accuracy UInt8,

    dribbles UInt8,

    fouls_drawn UInt8,
    fouls_commited UInt8,

    yellow_card UInt8,
    red_card UInt8
)
ENGINE = MergeTree()
ORDER BY (fixture_id, team_id, player_id);

CREATE TABLE fact_card_event
(
    event_id UInt32,

    fixture_id UInt32,
    team_id UInt32,
    player_id UInt32,
    player_name LowCardinality(String),
    minute UInt8,

    detail LowCardinality(String)
)
ENGINE = MergeTree()
ORDER BY (fixture_id, minute, event_id);


CREATE TABLE fact_substitution_event
(
    event_id UInt32,
    fixture_id UInt32,
    team_id UInt32,
    
    player_in_id UInt32,
    player_in_name String,
    player_out_id UInt32,
    player_out_name String,

    minute UInt8,

    detail String
)
ENGINE = MergeTree()
ORDER BY (fixture_id, team_id, minute, event_id);

CREATE TABLE fact_goal_events
(
    event_id UInt32,
    fixture_id UInt32,
    team_id UInt32,

    player_score_id UInt32,
    player_score_name String,

    player_assist_id Nullable(UInt32),
    player_assist_name Nullable(String),

    minute UInt8,

    detail LowCardinality(String)
)
ENGINE = MergeTree()
ORDER BY (fixture_id, team_id, minute, event_id);

CREATE TABLE fact_var_events
(
    event_id UInt32,
    fixture_id UInt32,
    team_id UInt32,

    player_id UInt32,
    player_name String,

    minute UInt8,

    detail LowCardinality(String)
)
ENGINE = MergeTree()
ORDER BY (fixture_id, team_id, minute, event_id);

CREATE TABLE head_to_head
(
    id UInt32,

    home_team_id UInt32,
    home_team_name String,

    away_team_id UInt32,
    away_team_name String,

    league_name String,

    home_goals UInt8,
    away_goals UInt8,

    match_date Date
)
ENGINE = MergeTree()
ORDER BY (
    home_team_id,
    away_team_id,
    match_date
);
